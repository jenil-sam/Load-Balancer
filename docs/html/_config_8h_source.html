66 bool load(const std::string &filename)
67 {
68 std::ifstream f(filename);
69 if (!f.is_open())

71 std::cerr << "Config file not found: " << filename
72 << " -- using defaults.\n";

75 std::string line;
76 while (std::getline(f, line))
77 {
78 if (line.empty() || line[0] == '#')
79 continue;
80 auto pos = line.find('=');
81 if (pos == std::string::npos)
82 continue;
83 std::string key = line.substr(0, pos);
84 std::string val = line.substr(pos + 1);
85 // Trim whitespace
86 key.erase(0, key.find_first_not_of(" \t"));
87 key.erase(key.find_last_not_of(" \t") + 1);

91 if (key == "servers")

102 newReqInterval = std::stoi(val);
103 else if (key == "use_switch")
104 useSwitch = (val == "1" || val == "true");
105 else if (key == "log_file")
106 logFile = val;
107 else if (key == "block_ip")
